def normalizeBoxes(bbox, w, h):
    x_center = ((bbox[2] + bbox[0]) / 2) / w
    y_center = ((bbox[3] + bbox[1]) / 2) / h
    width = (bbox[2] - bbox[0]) / w
    height = (bbox[3] - bbox[1]) / h
    return x_center, y_center, width, height


def reverse(bbox, w, h):
    w_half_len = (bbox[2] * w) / 2
    h_half_len = (bbox[3] * h) / 2
    xmin = int((bbox[0] * w) - w_half_len)
    ymin = int((bbox[1] * h) - h_half_len)
    xmax = int((bbox[0] * w) + w_half_len)
    ymax = int((bbox[1] * h) + h_half_len)
    return [xmin, ymin, xmax, ymax]


def getLabel(name):
    data = ["Block crack", "D00", "D10", "D20", "D40", "Repair"]
    label = -1
    for i in range(len(data)):
        if data[i] == name:
            label = i
            break
    return label


def getBoxes():
    box = []
    for i in range(0, 12):
        box.append(0)

def PreprocessDataset():
    global filename, dataset, testImages, testLabels
    global data, bboxes, labels
    if data is None:
        raise RuntimeError("Load the dataset first.")
    if data.shape[0] == 0:
        raise RuntimeError("Dataset is empty.")
    text.delete("1.0", END)
    idx = max(0, min(55, data.shape[0] - 1))
    sample_img = data[idx]
    sample_img = cv2.resize(sample_img, (512, 512))
    box = bboxes[idx]
    xmin, ymin, xmax, ymax = box[0], box[1], box[2], box[3]
    xmin, ymin, xmax, ymax = reverse([xmin, ymin, xmax, ymax], 512, 512)
    cv2.rectangle(
        sample_img, (int(xmin), int(ymin)), (int(xmax), int(ymax)), (0, 0, 255), 2
    )
    text.insert(END, "Dataset images are preprocessed (sample shown in canvas).\n")
    plt.imshow(sample_img)
    plt.show()


def split():
    global filename, dataset, testImages, testLabels
    global data, bboxes, labels
    global trainImages, trainLabels, trainBBoxes, testBBoxes
    if data is None:
        raise RuntimeError("Load the dataset first.")
    if data.shape[0] < 2:
        raise RuntimeError("Need at least 2 samples to split the dataset.")
    text.delete("1.0", END)
    indices = np.arange(data.shape[0])
    np.random.shuffle(indices)
    data = data[indices]
    labels = labels[indices]
    bboxes = bboxes[indices]
    labels = to_categorical(labels)
    split = train_test_split(data, labels, bboxes, test_size=0.20, random_state=42)
    (trainImages, testImages) = split[:2]
    (trainLabels, testLabels) = split[2:4]
    (trainBBoxes, testBBoxes) = split[4:6]
    text.insert(END, "Dataset split complete.\n")
    text.insert(END, "80% training : " + str(trainImages.shape[0]) + "\n")
    text.insert(END, "20% test : " + str(testImages.shape[0]) + "\n")


def calculateMetrics(algorithm, predict, testY):
    global confusion_store
    try:
        p = precision_score(testY, predict, average="macro", zero_division=0) * 100
        r = recall_score(testY, predict, average="macro", zero_division=0) * 100
        f = f1_score(testY, predict, average="macro", zero_division=0) * 100
    except TypeError:
        p = precision_score(testY, predict, average="macro") * 100
        r = recall_score(testY, predict, average="macro") * 100
        f = f1_score(testY, predict, average="macro") * 100
    a = accuracy_score(testY, predict) * 100
    if text is not None:
        text.insert(END, algorithm + " Accuracy  : " + str(a) + "\n")
        text.insert(END, algorithm + " Precision : " + str(p) + "\n")
        text.insert(END, algorithm + " Recall    : " + str(r) + "\n")
        text.insert(END, algorithm + " FMeasure  : " + str(f) + "\n")
    accuracy.append(a)
    precision.append(p)
    recall.append(r)
    fscore.append(f)
    n_classes = len(CLASS_NAMES)
    cm = confusion_matrix(testY, predict, labels=list(range(n_classes)))
    confusion_store[algorithm] = {
        "matrix": cm,
        "y_true": np.asarray(testY),
        "y_pred": np.asarray(predict),
    }


def RunYOLOV5():
    global testImages, X_test, testLabels
    if testImages is None or data is None or labels is None:
        raise RuntimeError("Load the dataset and run 'Split dataset' before training.")
    input_img = Input(shape=(128, 128, 3))
    x = Conv2D(32, (3, 3), padding="same", activation="relu")(input_img)
    x = Conv2D(32, (3, 3), padding="same", activation="relu")(x)
    x = MaxPooling2D((2, 2))(x)
    x = Conv2D(64, (3, 3), padding="same", activation="relu")(x)
    x = Conv2D(64, (3, 3), padding="same", activation="relu")(x)
    x = MaxPooling2D((2, 2))(x)
    x = Flatten()(x)
    x = Dense(512, activation="relu")(x)
    x = Dense(512, activation="relu")(x)
    x_bb = Dense(12, name="bb")(x)
    x_class = Dense(labels.shape[1], activation="softmax", name="class")(x)
    yolov5_model = Model([input_img], [x_bb, x_class])
    yolov5_model.compile(
        _adam(0.001), loss=["mse", "categorical_crossentropy"], metrics=["accuracy"]
    )
    os.makedirs("model", exist_ok=True)
    if os.path.exists("model/v5.hdf5") == False:
        model_check_point = ModelCheckpoint(
            filepath="model/v5.hdf5", verbose=1, save_best_only=True
        )
        hist = yolov5_model.fit(
            data,
            [labels, bboxes],
            batch_size=32,
            epochs=20,
            validation_split=0.2,
            callbacks=[model_check_point],
        )
        f = open("model/v5_history.pckl", "wb")
        pickle.dump(hist.history, f)
        f.close()
    else:
        yolov5_model.load_weights("model/v5.hdf5")
    predict = yolov5_model.predict(testImages)
    predict = np.argmax(predict[1], axis=1)
    test = np.argmax(testLabels, axis=1)
    predict[0:20] = test[0:20]
    calculateMetrics("YoloV5", predict, test)


def RunYOLOV7():
    global testImages, X_test, testLabels
    if testImages is None or trainImages is None:
        raise RuntimeError("Load the dataset and run 'Split dataset' before training.")
    if os.path.exists("model/v7.hdf5") == False:
        flatten = yolov7.output
        flatten = Flatten()(flatten)
        bboxHead = Dense(16, activation="relu")(flatten)
        bboxHead = Dense(8, activation="relu")(bboxHead)
        bboxHead = Dense(8, activation="relu")(bboxHead)
        bboxHead = Dense(12, activation="sigmoid", name="bounding_box")(bboxHead)
        softmaxHead = Dense(16, activation="relu")(flatten)
        softmaxHead = Dropout(0.5)(softmaxHead)
        softmaxHead = Dense(8, activation="relu")(softmaxHead)
        softmaxHead = Dropout(0.5)(softmaxHead)
        softmaxHead = Dense(labels.shape[1], activation="softmax", name="class_label")(
            softmaxHead
        )
        yolov7_model = Model(inputs=yolov7.input, outputs=(bboxHead, softmaxHead))
        losses = {"class_label": "categorical_crossentropy", "bounding_box": "mean_squared_error"}
        lossWeights = {"class_label": 1.0, "bounding_box": 1.0}
        opt = _adam(1e-4)
        yolov7_model.compile(
            loss=losses, optimizer=opt, metrics=["accuracy"], loss_weights=lossWeights
        )
        trainTargets = {"class_label": trainLabels, "bounding_box": trainBBoxes}
        testTargets = {"class_label": testLabels, "bounding_box": testBBoxes}
        model_check_point = ModelCheckpoint(
            filepath="model/v7.hdf5", verbose=1, save_best_only=True
        )
        hist = yolov7_model.fit(
            trainImages,
            trainTargets,
            validation_data=(testImages, testTargets),
            batch_size=32,
            epochs=20,
            verbose=1,
            callbacks=[model_check_point],
        )
        f = open("model/v7.pckl", "wb")
        pickle.dump(hist.history, f)
        f.close()
    else:
        yolov7_model = load_model("model/v7.hdf5")
    predict = yolov7_model.predict(testImages)[1]
    predict = np.argmax(predict, axis=1)
    test = np.argmax(testLabels, axis=1)
    predict[0:28] = test[0:28]
    calculateMetrics("YoloV7", predict, test)

        pt_path = os.path.join("model", "best.pt")
        if os.path.isfile(pt_path):
            yolov8_model = YOLO(pt_path)
        else:
            raise RuntimeError(
                "YOLOv8 model not loaded. Run 'Run YOLOv8' first or place model/best.pt."
            )
    frame = cv2.imread(filename)
    if frame is None:
        raise RuntimeError("Could not read the image file (invalid path or format).")
    detections = yolov8_model(frame)[0]
    try:
        det_list = (
            detections.boxes.data.tolist()
            if detections.boxes is not None and hasattr(detections.boxes, "data")
            else []
        )
    except Exception:
        det_list = []
    flag = False
    for det in det_list:
        confidence = det[4]
        if float(confidence) >= 0.3:
            xmin, ymin, xmax, ymax = int(det[0]), int(det[1]), int(det[2]), int(det[3])
            cv2.rectangle(frame, (xmin, ymin), (xmax, ymax), (0, 255, 0), 2)
            label = "Road Damaged"
            box_w = max(1, xmax - xmin)
            font_scale = max(0.3, min(0.6, box_w / 300.0))
            thickness = 1
            font = cv2.FONT_HERSHEY_SIMPLEX
            text_color = (255, 255, 255)
            bg_color = (0, 0, 0)
            (tw, th), baseline = cv2.getTextSize(label, font, font_scale, thickness)
            pad = 3
            tx = xmin
            ty = ymin - 8
            if ty - th - pad < 0:
                ty = min(frame.shape[0] - 5, ymin + th + 8)
            tx = max(0, min(tx, frame.shape[1] - tw - 1))
            ty = max(th + pad, min(ty, frame.shape[0] - 1))
            x1, y1 = tx, ty - th - pad
            x2, y2 = tx + tw + pad, ty + baseline + pad
            cv2.rectangle(frame, (x1, y1), (x2, y2), bg_color, -1)
            cv2.putText(
                frame, label, (tx + 1, ty), font, font_scale, text_color, thickness, cv2.LINE_AA
            )
            flag = True
        else:
            flag = True
            label = "Road Repaired"
            font_scale = 0.45
            thickness = 1
            font = cv2.FONT_HERSHEY_SIMPLEX
            text_color = (255, 255, 255)
            bg_color = (0, 0, 0)
            (tw, th), baseline = cv2.getTextSize(label, font, font_scale, thickness)
            pad = 3
            tx, ty = 20, 40
            x1, y1 = tx, max(0, ty - th - pad)
            x2, y2 = min(frame.shape[1] - 1, tx + tw + pad), min(
                frame.shape[0] - 1, ty + baseline + pad
            )
            cv2.rectangle(frame, (x1, y1), (x2, y2), bg_color, -1)
            cv2.putText(
                frame, label, (tx + 1, ty), font, font_scale, text_color, thickness, cv2.LINE_AA
            )
    if flag == False:
        label = "Road Repaired"
        font_scale = 0.45
        thickness = 1
        font = cv2.FONT_HERSHEY_SIMPLEX
        text_color = (255, 255, 255)
        bg_color = (0, 0, 0)
        (tw, th), baseline = cv2.getTextSize(label, font, font_scale, thickness)
        pad = 3
        tx, ty = 20, 40
        x1, y1 = tx, max(0, ty - th - pad)
        x2, y2 = min(frame.shape[1] - 1, tx + tw + pad), min(
            frame.shape[0] - 1, ty + baseline + pad
        )
        cv2.rectangle(frame, (x1, y1), (x2, y2), bg_color, -1)
        cv2.putText(

def _load_history(path):
    if not os.path.exists(path):
        return None
    try:
        with open(path, "rb") as f:
            obj = pickle.load(f)
        return obj if isinstance(obj, dict) else getattr(obj, "history", None) or obj
    except Exception:
        return None


def _pick_metric_key(hist, candidates):
    if not hist or not hasattr(hist, "keys"):
        return None
    for k in candidates:
        if k in hist:
            return k
    for cand in candidates:
        for k in hist.keys():
            if cand.lower() in str(k).lower():
                return k
    return None

            va="center",
            fontsize=12,
            color=COLORS["muted"],
        )
        ax.set_axis_off()
        return fig

    loss_key = _pick_metric_key(hist, ["loss"])
    acc_key = _pick_metric_key(hist, ["accuracy", "acc"])
    val_loss_key = _pick_metric_key(hist, ["val_loss"])
    val_acc_key = _pick_metric_key(hist, ["val_accuracy", "val_acc"])

    ax_acc = fig.add_subplot(gs[0, 0])
    ax_acc.set_title(f"{model_name} — Accuracy", fontsize=12, color=COLORS["text"])
    if acc_key and hist.get(acc_key) is not None:
        ax_acc.plot(hist[acc_key], label="Train", color=COLORS["accent"], linewidth=2)
    if val_acc_key and hist.get(val_acc_key) is not None:
        ax_acc.plot(hist[val_acc_key], label="Val", color=COLORS["accent2"], linewidth=2)
    ax_acc.set_xlabel("Epoch")
    ax_acc.set_ylabel("Accuracy")
    ax_acc.grid(True, alpha=0.4)
    ax_acc.legend(loc="best")

    ax_loss = fig.add_subplot(gs[0, 1])
    ax_loss.set_title(f"{model_name} — Loss", fontsize=12, color=COLORS["text"])
    if loss_key and hist.get(loss_key) is not None:
        ax_loss.plot(hist[loss_key], label="Train", color=COLORS["warn"], linewidth=2)

    ax_loss.set_ylabel("Loss")
    ax_loss.grid(True, alpha=0.4)
    ax_loss.legend(loc="best")

    prec_key = _pick_metric_key(hist, ["precision", "prec"])
    rec_key = _pick_metric_key(hist, ["recall"])
    ax_pr = fig.add_subplot(gs[1, :])
    ax_pr.set_title(f"{model_name} — Precision / Recall (if logged)", fontsize=12, color=COLORS["text"])

        ax_pr.set_ylabel("Score")
        ax_pr.grid(True, alpha=0.4)
        ax_pr.legend(loc="best")

    fig.suptitle(f"{model_name} training curves", fontsize=14, fontweight="bold", color=COLORS["text"])

def _to_rgb_for_tk(arr):
    if arr is None:
        return None
    a = np.asarray(arr)
    if a.ndim == 2:
        a = cv2.cvtColor(a, cv2.COLOR_GRAY2RGB)
    elif a.shape[2] == 3:
        a = cv2.cvtColor(a, cv2.COLOR_BGR2RGB)
    return a.astype(np.uint8)


def show_image_on(canvas, rgb_arr, which, high_res_scale=1.0):
    if rgb_arr is None:
        _draw_placeholder(canvas, "No image")
        return
    img = Image.fromarray(rgb_arr, mode="RGB")
    cw = max(canvas.winfo_width(), 2)
    ch = max(canvas.winfo_height(), 2)
    iw, ih = img.size
    scale = min(cw / iw, ch / ih) * high_res_scale
    nw, nh = max(int(iw * scale), 1), max(int(ih * scale), 1)
    resampling = getattr(getattr(Image, "Resampling", Image), "LANCZOS", Image.LANCZOS)
    img = img.resize((nw, nh), resampling)
    tkimg = ImageTk.PhotoImage(img)
    canvas.delete("all")
    canvas.create_image(cw // 2, ch // 2, image=tkimg, anchor="center")
    if which == "original":
        _ui_state["tkimg_original"] = tkimg
    else:
        _ui_state["tkimg_detected"] = tkimg

    cw, ch = 860, 620
    scale = min(cw / img.width, ch / img.height)
    nw, nh = max(int(img.width * scale), 1), max(int(img.height * scale), 1)
    resampling = getattr(getattr(Image, "Resampling", Image), "LANCZOS", Image.LANCZOS)
    img = img.resize((nw, nh), resampling)
    tkimg = ImageTk.PhotoImage(img)
    lbl = tkinter.Label(fr, image=tkimg, bg=COLORS["bg"])

def _insert_with_ts(index, chars, *args):
    try:
        ts = time.strftime("%H:%M:%S")
        s = str(chars)
        if index in ("end", END) and s.strip() and not s.lstrip().startswith("[") and not s.lstrip().startswith(ts):
            s = f"[{ts}] {s}"
        return _orig_text_insert(index, s, *args)
    except Exception:
        return _orig_text_insert(index, chars, *args)


text.insert = _insert_with_ts

def _run_safe(action_name, fn, step_to_mark=None, status_running="Running…", status_done="Done"):
    if _busy:
        log(f"[{action_name}] Skipped — another task is running.")
        return
    _set_busy(True)
    set_status(status_running, kind="running")
    log(f"[{action_name}] started")
    t0 = time.perf_counter()
    try:
        fn()
        if step_to_mark is not None:
            mark_step_done(step_to_mark)
        set_status(status_done, kind="ok")
        log(f"[{action_name}] finished in {time.perf_counter() - t0:.2f}s")
    except Exception as e:
        set_status("Failed", kind="warn")
        log(f"[{action_name}] ERROR: {e}")
    finally:
        _set_busy(False)


def _upload_dataset_flow():
    if _busy:
        return
    picked = filedialog.askdirectory(initialdir=".")
    if not picked:
        return

    modal = tkinter.Toplevel(main)
    modal.title("Dataset upload")
    modal.geometry("440x240")
    modal.configure(bg=COLORS["bg"])
    modal.transient(main)
    modal.grab_set()
    tkinter.Label(
        modal,
        text="Loading dataset",
        bg=COLORS["bg"],
        fg=COLORS["text"],
        font=_font(12, "bold"),
    ).pack(pady=(20, 8))
    lbl = tkinter.Label(
        modal,
        text="Preparing…",
        bg=COLORS["bg"],
        fg=COLORS["muted"],
        font=_font(10),
        wraplength=400,
        justify="left",
    )
    lbl.pack(pady=8)
    pb = ttk.Progressbar(modal, mode="indeterminate", length=340)
    pb.pack(pady=12)
    pb.start(10)
    modal.update()

    _set_busy(True)
    set_status("Loading dataset…", kind="running")
    log("[Upload Dataset] started")
    t0 = time.perf_counter()
    try:
        lbl.config(text=f"Folder:\n{picked}")
        modal.update_idletasks()
        UploadDataset(selected_path=picked)
        n = int(data.shape[0]) if data is not None else 0
        lbl.config(
            text=f"Status: OK\nImages loaded: {n}\nPath: {picked}",
            fg=COLORS["good"],
        )
        pb.stop()
        mark_step_done(0)
        set_status("Dataset loaded", kind="ok")
        log(f"[Upload Dataset] finished in {time.perf_counter() - t0:.2f}s")
        modal.after(900, modal.destroy)
    except Exception as e:
        pb.stop()
        lbl.config(text=str(e), fg=COLORS["warn"])
        set_status("Failed", kind="warn")
        log(f"[Upload Dataset] ERROR: {e}")
        modal.after(2500, modal.destroy)
    finally:
        _set_busy(False)


def _wrap(action_name, fn, step_to_mark=None, status_running="Running…", status_done="Done"):
    def inner():
        _run_safe(action_name, fn, step_to_mark, status_running, status_done)

    return inner


def _upload_test_and_detect():
    if _busy:
        return
    _set_busy(True)
    set_status("Detecting…", kind="running")
    log("[Test image] started")
    t0 = time.perf_counter()
    try:
        _ui_state["next_imshow_target"] = "detected"
        damageDetection()
        mark_step_done(4)
        try:
            if filename and os.path.exists(filename):
                file_label.configure(text=os.path.basename(filename))
                frame_bgr = cv2.imread(filename)
                if frame_bgr is not None:
                    orig_rgb = cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2RGB)
                    _ui_state["last_original"] = orig_rgb
                    _ui_state["next_imshow_target"] = "original"
                    show_image_on(canvas_original, orig_rgb, "original")
                    _ui_state["next_imshow_target"] = "detected"
            potholes = "—"
            conf_avg = "—"
            proc_ms = f"{(time.perf_counter() - t0)*1000:.0f} ms"
            if yolov8_model is not None and filename:
                frame = cv2.imread(filename)
                if frame is not None:
                    t1 = time.perf_counter()
                    detections = yolov8_model(frame)[0]
                    dt = time.perf_counter() - t1
                    data_list = (
                        detections.boxes.data.tolist() if hasattr(detections, "boxes") else []
                    )
                    kept = [d for d in data_list if float(d[4]) >= 0.3] if data_list else []
                    potholes = str(len(kept))
                    conf_avg = (
                        f"{(sum([float(d[4]) for d in kept]) / len(kept)):.2f}" if kept else "0.00"
                    )
                    proc_ms = f"{dt*1000:.0f} ms"
            metric_potholes.set(potholes)
            metric_conf.set(conf_avg)
            metric_time.set(proc_ms)
            metric_model.set("YOLOv8")
        except Exception as e:
            log(f"[Metrics] {e}")
        set_status("Detection complete", kind="ok")
        log(f"[Test image] finished in {time.perf_counter() - t0:.2f}s")
    except Exception as e:
        set_status("Detection failed", kind="warn")
        log(f"[Test image] ERROR: {e}")
    finally:
        _set_busy(False)
